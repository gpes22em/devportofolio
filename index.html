-- ============================================================ -- PostgreSQL Database Setup untuk Vercel + Neon (atau Vercel Postgres) -- ============================================================ -- Jika pakai Neon/Supabase: CREATE DATABASE tidak perlu (sudah dibuat di dashboard) -- Langsung jalankan query CREATE TABLE saja. -- Buat tabel orders (PostgreSQL syntax) CREATE TABLE IF NOT EXISTS orders ( id SERIAL PRIMARY KEY, name VARCHAR(255) NOT NULL, whatsapp VARCHAR(50) NOT NULL, category VARCHAR(100) NOT NULL, budget VARCHAR(100) NOT NULL, description TEXT NOT NULL, ip_address VARCHAR(45) DEFAULT NULL, status VARCHAR(20) DEFAULT 'new', wa_notified INTEGER DEFAULT 0, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); -- Index untuk pencarian cepat CREATE INDEX IF NOT EXISTS idx_orders_status ON orders(status); CREATE INDEX IF NOT EXISTS idx_orders_created ON orders(created_at);
